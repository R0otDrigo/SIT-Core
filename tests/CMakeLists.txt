# Cada integrante agrega su prueba como tests/test_<tema>.cpp: no hace falta editar este archivo.
# Las pruebas test_gui_*.cpp usan Qt y solo se compilan si Qt6 esta instalado.
file(GLOB SIT_TEST_SOURCES CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test_*.cpp)

foreach(src ${SIT_TEST_SOURCES})
  get_filename_component(t ${src} NAME_WE)
  if(t MATCHES "^test_gui_")
    if(NOT TARGET sit_ui)
      continue()
    endif()
    add_executable(${t} ${src})
    target_link_libraries(${t} PRIVATE sit_ui)
  else()
    add_executable(${t} ${src})
    target_link_libraries(${t} PRIVATE sit_model)
  endif()
  target_compile_definitions(${t} PRIVATE SIT_DATA_DIR="${PROJECT_SOURCE_DIR}/data")
  add_test(NAME ${t} COMMAND ${t})
  if(t MATCHES "^test_gui_")
    set_tests_properties(${t} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  endif()
endforeach()
